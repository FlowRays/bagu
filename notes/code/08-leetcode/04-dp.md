## 股票系列

### LC121. 买卖股票的最佳时机 ⭐

**思路**：维护历史最低价，边扫边更新最大差。

```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int minPrice = INT_MAX, ans = 0;
        for (int p : prices) {
            minPrice = min(minPrice, p);       // 维护历史最低价
            ans = max(ans, p - minPrice);
        }
        return ans;
    }
};
```

**复杂度**：O(n)

**易错**：只能交易一次，所以是「最低价之后的最高价」，不是全局 max-min。

### LC122. 买卖股票的最佳时机 II ⭐

**思路**：所有上涨区间全部吃掉。

```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int ans = 0;
        for (size_t i = 1; i < prices.size(); ++i)
            ans += max(0, prices[i] - prices[i-1]);   // 所有上涨区间全吃掉
        return ans;
    }
};
```

**复杂度**：O(n)

**易错**：`ans += max(0, p[i]-p[i-1])`；不需要真的模拟买卖点。

### LC123. 买卖股票的最佳时机 III

**思路**：四个状态滚动：buy1 / sell1 / buy2 / sell2。

```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        // buy1: 第一次买入后的最大收益；sell1: 第一次卖出后；以此类推
        int buy1 = INT_MIN, sell1 = 0, buy2 = INT_MIN, sell2 = 0;
        for (int p : prices) {
            buy1  = max(buy1,  -p);
            sell1 = max(sell1, buy1 + p);
            buy2  = max(buy2,  sell1 - p);
            sell2 = max(sell2, buy2 + p);
        }
        return sell2;
    }
};
```

**复杂度**：O(n) / O(1)

**易错**：**顺序不能乱**，同一天内 buy2 依赖 sell1，写在一个循环里正好是允许当天买卖的等价形式；buy 初值 `INT_MIN`。

### LC188. 买卖股票的最佳时机 IV

**思路**：k 次交易的通用 DP：`buy[j] / sell[j]`。

```cpp
class Solution {
public:
    int maxProfit(int k, vector<int>& prices) {
        int n = prices.size();
        if (n == 0 || k == 0) return 0;
        if (k >= n / 2) {                        // 次数够多，退化成 LC122
            int ans = 0;
            for (int i = 1; i < n; ++i) ans += max(0, prices[i] - prices[i-1]);
            return ans;
        }
        // buy[j]: 已完成 j-1 次交易并处于持有状态的最大收益
        vector<int> buy(k + 1, INT_MIN), sell(k + 1, 0);
        for (int p : prices)
            for (int j = 1; j <= k; ++j) {
                buy[j]  = max(buy[j],  sell[j-1] - p);
                sell[j] = max(sell[j], buy[j] + p);
            }
        return sell[k];
    }
};
```

**复杂度**：O(n·k)

**易错**：`k >= n/2` 要退化成 LC122，否则 k 很大时会 MLE/TLE；内层 j 从 1 到 k，`buy[j]` 依赖 `sell[j-1]`。

## 一维 DP

### LC139. 单词拆分

**思路**：`dp[i]` 表示前 i 个字符可拆。枚举分割点 j。

```cpp
class Solution {
public:
    bool wordBreak(string s, vector<string>& wordDict) {
        unordered_set<string> dict(wordDict.begin(), wordDict.end());
        int n = s.size();
        vector<char> dp(n + 1, false);
        dp[0] = true;                              // 空串可拆
        for (int i = 1; i <= n; ++i)
            for (int j = 0; j < i; ++j)
                if (dp[j] && dict.count(s.substr(j, i - j))) { dp[i] = true; break; }
        return dp[n];
    }
};
```

**复杂度**：O(n²·L)

**易错**：`dp[0] = true`；字典用 `unordered_set`；`substr(j, i-j)` 的长度参数别写成 `i`。

### LC300. 最长递增子序列 ⭐

**思路**：贪心 + 二分：`tails[i]` = 长度 i+1 的递增子序列的**最小结尾**。

```cpp
// O(n^2) DP
class SolutionDP {
public:
    int lengthOfLIS(vector<int>& nums) {
        if (nums.empty()) return 0;
        vector<int> dp(nums.size(), 1);
        int ans = 1;
        for (size_t i = 1; i < nums.size(); ++i) {
            for (size_t j = 0; j < i; ++j)
                if (nums[j] < nums[i]) dp[i] = max(dp[i], dp[j] + 1);
            ans = max(ans, dp[i]);
        }
        return ans;
    }
};
// O(n log n)：tails[i] = 长度为 i+1 的递增子序列的最小结尾
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        vector<int> tails;
        for (int x : nums) {
            auto it = lower_bound(tails.begin(), tails.end(), x);  // 严格递增用 lower_bound
            if (it == tails.end()) tails.push_back(x);
            else *it = x;
        }
        return tails.size();                       // 注意 tails 本身不是某个 LIS
    }
};
```

**复杂度**：O(n log n)

**易错**：严格递增用 `lower_bound`，非严格用 `upper_bound`；**`tails` 本身不是任何一个合法 LIS**，只有长度有意义。O(n²) 的 DP 版也要会写。

## 二维 DP

### LC64. 最小路径和

**思路**：`dp[j] = min(上, 左) + grid[i][j]`，滚动成一维。

```cpp
class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        int m = grid.size(), n = grid[0].size();
        vector<int> dp(n, 0);                      // 滚动一维
        dp[0] = grid[0][0];
        for (int j = 1; j < n; ++j) dp[j] = dp[j-1] + grid[0][j];
        for (int i = 1; i < m; ++i) {
            dp[0] += grid[i][0];
            for (int j = 1; j < n; ++j)
                dp[j] = min(dp[j], dp[j-1]) + grid[i][j];   // dp[j] 是上方，dp[j-1] 是左方
        }
        return dp[n-1];
    }
};
```

**复杂度**：O(m·n) / O(n)

**易错**：滚动时 `dp[j]` 是上方（还没更新）、`dp[j-1]` 是左方（已更新），顺序天然正确；第一行第一列要单独初始化。

### LC1143. 最长公共子序列 ⭐

**思路**：`dp[i][j]`：a 前 i 个和 b 前 j 个的 LCS。相等 `+1`，否则取两边 max。

```cpp
class Solution {
public:
    int longestCommonSubsequence(string a, string b) {
        int m = a.size(), n = b.size();
        vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
        for (int i = 1; i <= m; ++i)
            for (int j = 1; j <= n; ++j)
                dp[i][j] = (a[i-1] == b[j-1]) ? dp[i-1][j-1] + 1
                                              : max(dp[i-1][j], dp[i][j-1]);
        return dp[m][n];
    }
};
```

**复杂度**：O(m·n)

**易错**：下标偏移 1，`a[i-1]` 对应 `dp[i]`；LCS 和「最长公共子串」不是一回事（后者不相等时要清零）。

### LC72. 编辑距离 ⭐

**思路**：和 LCS 同一个骨架，转移多了插入/删除/替换三选一。

```cpp
class Solution {
public:
    int minDistance(string a, string b) {
        int m = a.size(), n = b.size();
        vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
        for (int i = 0; i <= m; ++i) dp[i][0] = i;      // 全删
        for (int j = 0; j <= n; ++j) dp[0][j] = j;      // 全插
        for (int i = 1; i <= m; ++i)
            for (int j = 1; j <= n; ++j)
                dp[i][j] = (a[i-1] == b[j-1])
                    ? dp[i-1][j-1]
                    : 1 + min({dp[i-1][j],      // 删 a[i-1]
                               dp[i][j-1],      // 插 b[j-1]
                               dp[i-1][j-1]});  // 替换
        return dp[m][n];
    }
};
```

**复杂度**：O(m·n)

**易错**：边界 `dp[i][0]=i`、`dp[0][j]=j` 必须初始化；相等时是 `dp[i-1][j-1]` **不加 1**。

### LC5. 最长回文子串

**思路**：中心扩散，奇偶两种中心各扫一遍。

```cpp
class Solution {
public:
    string longestPalindrome(string s) {
        if (s.empty()) return "";
        int start = 0, len = 1;
        auto expand = [&](int l, int r) {           // 中心扩散
            while (l >= 0 && r < (int)s.size() && s[l] == s[r]) { --l; ++r; }
            if (r - l - 1 > len) { len = r - l - 1; start = l + 1; }
        };
        for (int i = 0; i < (int)s.size(); ++i) {
            expand(i, i);        // 奇数长度
            expand(i, i + 1);    // 偶数长度
        }
        return s.substr(start, len);
    }
};
```

**复杂度**：O(n²) / O(1)

**易错**：偶数中心是 `expand(i, i+1)`，容易漏；退出循环时 `l`、`r` 都多走了一步，长度是 `r-l-1`、起点是 `l+1`。
